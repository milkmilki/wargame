extends SceneTree
const Codec = preload("res://.dbg/state_codec.gd")
func _initialize(): call_deferred("run")
func run():
	var expected: Dictionary = JSON.parse_string(FileAccess.open("res://.dbg/supply-current-day60.json",FileAccess.READ).get_as_text())
	var hashes := {}
	for path in ["scripts/simulation/simulation.gd","scripts/simulation/rules/supply_rules.gd","scripts/core/pathfinding.gd","tests/support/legacy_supply_batch_simulation.gd"]: hashes[path]=FileAccess.get_sha256("res://"+path)
	var results := {}
	var variants := ["before","after"] if OS.get_process_id()%2==0 else ["after","before"]
	for variant in variants:
		var sim: Simulation = Codec.new().restore("res://.dbg/supply-current-day60.bin","res://tests/support/legacy_supply_batch_simulation.gd" if variant=="before" else "")
		root.add_child(sim); sim.paused=true
		var digest := HashingContext.new(); digest.start(HashingContext.HASH_SHA256); digest.update(var_to_bytes(NativeSnapshotBuilder.build(sim.state)))
		assert(digest.finish().hex_encode()==expected.sha256,"identical current Eurasia state")
		var prepare := []; var sources := []
		for repeat in range(30):
			var started := Time.get_ticks_usec(); var ids := sim._prepare_supply_network_caches(); prepare.append(Time.get_ticks_usec()-started)
			await sim._prebuild_supply_networks_over_frames(ids)
			started=Time.get_ticks_usec(); var result: Dictionary = await sim._precompute_supply_sources_over_frames(); sources.append(Time.get_ticks_usec()-started)
			for army in sim.state.armies:
				if not result.has(army.get_instance_id()): continue
				var oracle: Array[Dictionary] = Pathfinding.supply_sources_from_network(sim.state,army,sim._daily_supply_network_cache.get(army.owner_nation,[] as Array[Dictionary]))
				assert(var_to_bytes(result[army.get_instance_id()])==var_to_bytes(oracle),"exact sources/order")
		var rows := []; sim.tick_phase_profiling_enabled=true
		for day in range(35):
			var started := Time.get_ticks_usec(); sim._advance_day(false); var elapsed := Time.get_ticks_usec()-started
			digest=HashingContext.new(); digest.start(HashingContext.HASH_SHA256); digest.update(var_to_bytes(NativeSnapshotBuilder.build(sim.state)))
			rows.append({"day":sim.state.day,"elapsed_usec":elapsed,"profile":sim.tick_profile_last_usec.duplicate(),"sha256":digest.finish().hex_encode()})
			if day%10==0: print("SUPPLY_COMPARE_DAY ",variant," ",sim.state.day); await process_frame
		results[variant]={"prepare_usec":prepare,"source_batch_wall_usec":sources,"daily":rows}
		sim.free(); await process_frame
	for index in range(35): assert(results.before.daily[index].sha256==results.after.daily[index].sha256,"identical full daily snapshot")
	FileAccess.open("res://.dbg/supply-compare-%d.json"%OS.get_process_id(),FileAccess.WRITE).store_string(JSON.stringify({"pid":OS.get_process_id(),"initial":expected.sha256,"mask":expected.mask,"systems":expected.systems,"seed":1,"days":[61,95],"hashes":hashes,"results":results},"  "))
	print("SUPPLY_COMPARE_COMPLETE ",OS.get_process_id()); quit()
