from pathlib import Path
import re, subprocess

def head(path):
    return subprocess.check_output(['git','show','HEAD:'+path]).decode('utf8').replace('\r\n','\n')
def stage(path,text):
    blob=subprocess.check_output(['git','hash-object','-w','--path='+path,'--stdin'],input=text.encode('utf8')).decode().strip()
    subprocess.run(['git','update-index','--cacheinfo','100644,'+blob+','+path],check=True)
def replace(text,old,new):
    assert text.count(old)==1,old
    return text.replace(old,new,1)
