from pathlib import Path
p=Path('.dbg/v19_modifier_bench.gd')
p.write_text('''extends SceneTree
const Reference = preload("res://tests/support/ruler_modifiers_reference.gd")
func _initialize(): call_deferred("run")
func measure(legacy: bool,profiles: Array[Nation]) -> Dictionary:
	var started := Time.get_ticks_usec(); var sum := 0.0
	for iteration in range(100000):
		var profile: Nation = profiles[iteration%profiles.size()]
		var values := Reference.modifiers(profile) if legacy else RulerProfile.modifiers(profile)
		sum += float(values[RulerProfile.KEY_FOOD_CONSUMPTION])
	return {"us":Time.get_ticks_usec()-started,"checksum":sum}
func run():
	var profiles: Array[Nation] = []
	for i in range(30):
		var nation := Nation.new();nation.ruler_archetype=RulerProfile.ARCHETYPE_IDS[i%RulerProfile.ARCHETYPE_IDS.size()]
		if i%3>0: nation.ruler_traits.append(RulerProfile.TRAIT_IDS[i%RulerProfile.TRAIT_IDS.size()])
		if i%3>1: nation.ruler_traits.append(RulerProfile.TRAIT_IDS[(i+5)%RulerProfile.TRAIT_IDS.size()])
		profiles.append(nation)
	var reverse := OS.get_cmdline_user_args().has("--reverse")
	var first := measure(reverse,profiles);var second := measure(not reverse,profiles)
	assert(first.checksum==second.checksum)
	var start := Time.get_ticks_usec(); var table := RulerProfile._build_combined_modifiers();var init_us := Time.get_ticks_usec()-start
	var output := {"pid":OS.get_process_id(),"before":first if reverse else second,"after":second if reverse else first,"entries":table.size(),"build_us":init_us,"serialized_bytes":var_to_bytes(table).size()}
	FileAccess.open("res://.dbg/ruler-table-v19/modifiers-%d.json"%OS.get_process_id(),FileAccess.WRITE).store_string(JSON.stringify(output,"  "))
	print("MODIFIER_BENCH_COMPLETE ",OS.get_process_id());quit()
''',encoding='utf-8')
import subprocess,json,statistics
out=Path('.dbg/ruler-table-v19'); results=[]
for run in range(5):
 with (out/f'modifier-{run}.stdout.txt').open('wb') as o,(out/f'modifier-{run}.stderr.txt').open('wb') as e:
  process=subprocess.Popen(['D:/ProjectICreate/Godot_v4.7.1-stable_win64.exe','--headless','--path',str(Path.cwd()),'--script','res://.dbg/v19_modifier_bench.gd','--']+(['--reverse'] if run%2 else []),stdout=o,stderr=e,creationflags=subprocess.CREATE_NO_WINDOW)
  code=process.wait(timeout=60)
  assert code==0 and not (out/f'modifier-{run}.stderr.txt').read_text(encoding='utf-8').strip()
  results.append(json.loads((out/f'modifiers-{process.pid}.json').read_text(encoding='utf-8')))
(out/'modifier-bench-summary.json').write_text(json.dumps(results,indent=2)+'\n',encoding='utf-8')
print('MODIFIER_BENCH',statistics.median(x['before']['us'] for x in results)/1000,statistics.median(x['after']['us'] for x in results)/1000,'entries',results[0]['entries'],'init_ms',statistics.median(x['build_us'] for x in results)/1000)
